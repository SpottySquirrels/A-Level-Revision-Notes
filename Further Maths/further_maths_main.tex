% Vector in ijk form (ai+bj+ck)
\newcommand{\ijkvec}[3]{%
    \def\ijkhasitem{0}%
    % i component
    \ifblank{#1}{}{%
        \ifnum#1=0 \relax
        \else
            \ifnum#1=1 \mathbf{i}%
            \else\ifnum#1=-1 -\mathbf{i}%
            \else #1\mathbf{i}%
            \fi\fi
            \def\ijkhasitem{1}%
        \fi
    }%
    % j component
    \ifblank{#2}{}{%
        \ifnum#2=0 \relax
        \else
            \ifnum#2>0 %
                \ifnum\ijkhasitem=1 +%
                \fi
            \fi
            \ifnum#2=1 \mathbf{j}%
            \else\ifnum#2=-1 -\mathbf{j}%
            \else #2\mathbf{j}%
            \fi\fi
            \def\ijkhasitem{1}%
        \fi
    }%
    % k component
    \ifblank{#3}{}{%
        \ifnum#3=0 \relax
        \else
            \ifnum#3>0 %
                \ifnum\ijkhasitem=1 +%
                \fi
            \fi
            \ifnum#3=1 \mathbf{k}%
            \else\ifnum#3=-1 -\mathbf{k}%
            \else #3\mathbf{k}%
            \fi\fi
        \fi
    }%
}
% ijk vector with non-numeric inputs
\newcommand{\ijkexprvec}[3]{
    % i component
    \ifblank{#1}{}{
        #1\mathbf{i}
    }
    % j component
    \ifblank{#2}{}{
        #2\mathbf{j}
    }
    \ifblank{#3}{}{
        #3\mathbf{k}
    }
}
